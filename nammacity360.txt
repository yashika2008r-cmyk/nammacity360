    // Upvote System Counter
    function upvoteReport(id) {
      const report = reports.find(r =&gt; r.id === id);
      if (!report) return;

      if (!report.hasUpvoted) {
        report.upvotes += 1;
        report.hasUpvoted = true;
      } else {
        report.upvotes -= 1;
        report.hasUpvoted = false;
      }

      updateStatsCounters();
      renderFeed();
      renderMapMarkers();
      
      if (mapMarkers[id] &amp;&amp; mapMarkers[id].isPopupOpen()) {
        mapMarkers[id].setPopupContent(buildPopupHTML(report));
      }
    }

      document.getElementById(&#x27;stat-resolved&#x27;).innerText = resolved;

    // Upvote System Counter
    function upvoteReport(id) {
      const report = reports.find(r =&gt; r.id === id);
      if (!report) return;

      if (!report.hasUpvoted) {
        report.upvotes += 1;
        report.hasUpvoted = true;
      } else {
        report.upvotes -= 1;
        report.hasUpvoted = false;
      }

      updateStatsCounters();
      renderFeed();
      renderMapMarkers();
      
      if (mapMarkers[id] &amp;&amp; mapMarkers[id].isPopupOpen()) {
        mapMarkers[id].setPopupContent(buildPopupHTML(report));
      }
    }

function straightDistanceKm(a,b){
 const rad=v=&gt;v*Math.PI/180;
 const dLat=rad(b[0]-a[0]),dLon=rad(b[1]-a[1]);
 const x=Math.sin(dLat/2)**2+Math.cos(rad(a[0]))*Math.cos(rad(b[0]))*Math.sin(dLon/2)**2;
 return 6371*2*Math.atan2(Math.sqrt(x),Math.sqrt(1-x));
}

 async function geocode(query){
  const url="https://nominatim.openstreetmap.org/search?format=jsonv2&limit=1&q="+encodeURIComponent(query);
  const res=await fetch(url,{headers:{"Accept":"application/json"}});
  if(!res.ok) throw new Error("Location search service is temporarily unavailable.");
  const data=await res.json();
  if(!data.length) throw new Error("Could not find: "+query+". Try a nearby landmark or full address.");
  return {lat:Number(data[0].lat),lng:Number(data[0].lon),label:data[0].display_name};
 }
 async function drawRoute(a,b){
  await init();
  if(!map) return;
  status("Finding both places and calculating the route…");
  const url=`https://router.project-osrm.org/route/v1/driving/${a.lng},${a.lat};${b.lng},${b.lat}?overview=full&geometries=geojson`;
  let data=null;
  try{const res=await fetch(url);if(res.ok)data=await res.json();}catch(e){}
  if(routeLayer) map.removeLayer(routeLayer);
  endpointLayer.clearLayers();
  const hasRoadRoute=data?.code==="Ok"&&data.routes?.length;
  const coords=hasRoadRoute?data.routes[0].geometry.coordinates.map(p=>[p[1],p[0]]):[[a.lat,a.lng],[((a.lat+b.lat)/2)+0.002,((a.lng+b.lng)/2)-0.001],[b.lat,b.lng]];
  routeLayer=L.polyline(coords,{color:"#087f68",weight:6,opacity:.9,dashArray:hasRoadRoute?null:"10 7"}).addTo(map);
  L.marker([a.lat,a.lng]).addTo(endpointLayer).bindPopup(`<b>START</b><br>${esc(a.label)}`).openPopup();

  map.fitBounds(routeLayer.getBounds(),{padding:[35,35]});
  if(hasRoadRoute){
   const km=(data.routes[0].distance/1000).toFixed(1), mins=Math.round(data.routes[0].duration/60);
